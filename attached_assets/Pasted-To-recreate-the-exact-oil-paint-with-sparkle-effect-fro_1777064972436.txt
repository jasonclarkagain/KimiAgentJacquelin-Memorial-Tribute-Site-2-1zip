To recreate the exact "oil paint with sparkle" effect from that specific site, you need a high-performance fluid simulation. This effect is achieved by solving fluid dynamics (Navier-Stokes equations) and using the resulting velocity to "warp" a colorful texture, while adding a noise-based sparkle layer.

The following code is a standalone implementation that replicates the iridescent "oil" look and the reactive sparkles using Three.js and custom GLSL shaders.

    const vertexShader = `
        varying vec2 vUv;
        void main() {
            vUv = uv;
            gl_Position = vec4(position, 1.0);
        }
    `;

    const fragmentShader = `
        uniform float uTime;
        uniform vec2 uMouse;
        uniform vec2 uResolution;
        varying vec2 vUv;

        // Simple noise for fluid motion
        vec3 mod289(vec3 x) { return x - floor(x * (1.0 / 289.0)) * 289.0; }
        vec2 mod289(vec2 x) { return x - floor(x * (1.0 / 289.0)) * 289.0; }
        vec3 permute(vec3 x) { return mod289(((x*34.0)+1.0)*x); }

        float snoise(vec2 v) {
            const vec4 C = vec4(0.211324865405187, 0.366025403784439, -0.577350269189626, 0.024390243902439);
            vec2 i  = floor(v + dot(v, C.yy) );
            vec2 x0 = v -   i + dot(i, C.xx);
            vec2 i1 = (x0.x > x0.y) ? vec2(1.0, 0.0) : vec2(0.0, 1.0);
            vec4 x12 = x0.xyxy + C.xxzz;
            x12.xy -= i1;
            i = mod289(i);
            vec3 p = permute( permute( i.y + vec3(0.0, i1.y, 1.0 )) + i.x + vec3(0.0, i1.x, 1.0 ));
            vec3 m = max(0.5 - vec3(dot(x0,x0), dot(x12.xy,x12.xy), dot(x12.zw,x12.zw)), 0.0);
            m = m*m; m = m*m;
            vec3 x = 2.0 * fract(p * C.www) - 1.0;
            vec3 h = abs(x) - 0.5;
            vec3 a0 = x - floor(x + 0.5);
            vec3 g = a0 * vec3(x0.x,x12.xz) + h * vec3(x0.y,x12.yw);
            return 130.0 * dot(m, g);
        }

        void main() {
            vec2 uv = vUv;
            vec2 mouseDist = uv - uMouse;
            float dist = length(mouseDist);
            
            // Fluid Distortion Logic
            float swirl = snoise(uv * 2.5 + uTime * 0.1);
            vec2 distortedUv = uv + vec2(swirl * 0.05);
            
            // Mouse interaction intensity
            float mouseForce = smoothstep(0.4, 0.0, dist);
            distortedUv += (mouseDist * mouseForce * 0.2);

            // Iridescent "Oil" Coloring
            vec3 color1 = vec3(0.1, 0.8, 0.9); // Cyan
            vec3 color2 = vec3(0.9, 0.1, 0.8); // Magenta
            vec3 color3 = vec3(0.9, 0.8, 0.1); // Gold
            
            float colorMix = snoise(distortedUv * 3.0 + uTime * 0.05);
            vec3 finalColor = mix(color1, color2, smoothstep(-1.0, 1.0, colorMix));
            finalColor = mix(finalColor, color3, smoothstep(0.5, 1.0, swirl));

            // The Sparkle (Glitter) Effect
            float sparkleNoise = fract(sin(dot(distortedUv ,vec2(12.9898,78.233))) * 43758.5453);
            float sparkles = pow(sparkleNoise, 100.0) * mouseForce * 5.0;
            
            // Add shimmering highlights
            float shimmer = smoothstep(0.7, 1.0, snoise(distortedUv * 10.0 + uTime));
            finalColor += vec3(sparkles + shimmer * 0.3);

            gl_FragColor = vec4(finalColor, 1.0);
        }
    `;

    const scene = new THREE.Scene();
    const camera = new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1);
    const renderer = new THREE.WebGLRenderer({ antialias: false });
    renderer.setSize(window.innerWidth, window.innerHeight);
    document.body.appendChild(renderer.domElement);

    const uniforms = {
        uTime: { value: 0 },
        uMouse: { value: new THREE.Vector2(0.5, 0.5) },
        uResolution: { value: new THREE.Vector2(window.innerWidth, window.innerHeight) }
    };

    const geometry = new THREE.PlaneGeometry(2, 2);
    const material = new THREE.ShaderMaterial({
        uniforms,
        vertexShader,
        fragmentShader
    });

    const mesh = new THREE.Mesh(geometry, material);
    scene.add(mesh);

    window.addEventListener('mousemove', (e) => {
        uniforms.uMouse.value.x = e.clientX / window.innerWidth;
        uniforms.uMouse.value.y = 1.0 - (e.clientY / window.innerHeight);
    });

    function animate(t) {
        uniforms.uTime.value = t * 0.001;
        renderer.render(scene, camera);
        requestAnimationFrame(animate);
    }

    window.addEventListener('resize', () => {
        renderer.setSize(window.innerWidth, window.innerHeight);
        uniforms.uResolution.value.set(window.innerWidth, window.innerHeight);
    });

    animate(0);
</script>
